[x] JS-01: Hoisting and Execution Context
--------------------------------------------------------------------------------
💡 THEORY:
- Execution Context: JavaScript executes code in an execution context. Each context
  has two phases: 
  1. Creation Phase (Memory Allocation): JavaScript scans code and sets up memory
     space for variables and functions. 
     - Variables declared with 'var' are initialized to 'undefined'.
     - Variables declared with 'let' and 'const' are allocated space but NOT 
       initialized (this creates the Temporal Dead Zone - TDZ).
     - Function declarations are hoisted with their complete function definition.
  2. Execution Phase: Code runs line-by-line; assignments are evaluated.

💻 CODING / OUTPUT QUESTIONS:

Question 1.1: What is the output of the following code?
-----------------------------------------
console.log(a); // undefined (var is hoisted and initialized to undefined)
var a = 10;
console.log(a); // 10

Question 1.2: What happens when let/const are accessed before declaration?
-----------------------------------------
console.log(b); // ReferenceError: Cannot access 'b' before initialization
let b = 20;

Question 1.3: Function Declaration vs Function Expression Hoisting
-----------------------------------------
greet(); // "Hello!" (Function declaration is hoisted fully)
function greet() {
  console.log("Hello!");
}

sayHello(); // TypeError: sayHello is not a function
var sayHello = function() {
  console.log("Hi!");
};
// Explanation: 'var sayHello' is hoisted as 'undefined'. Attempting to call
// undefined as a function throws a TypeError.

Question 1.4: Shadowing and Function-scoped Hoisting
-----------------------------------------
var x = 1;
function test() {
  if (!x) {
    var x = 100;
  }
  console.log(x); // Prints: 100
}
test();
// Explanation: Inside test(), 'var x' is hoisted to the top of the function scope.
// It shadows the global 'x' and has an initial value of 'undefined'. 
// Therefore, '!x' (!undefined) becomes true, assigning 100 to 'x'.

[x] JS-02: Scopes and Closures
--------------------------------------------------------------------------------
💡 THEORY:
- Lexical Scope: A function can access variables defined in its outer scopes.
- Closure: The combination of a function bundled together with references to its 
  lexical environment. It allows an inner function to remember and access its 
  outer variables even after the outer function has finished executing.

💻 CODING / PRACTICE PROBLEMS:

Problem 2.1: Implement a private counter using closures.
-----------------------------------------
function createCounter() {
  let count = 0; // Private variable (not accessible from outside)
  
  return {
    increment() {
      count++;
      return count;
    },
    decrement() {
      count--;
      return count;
    },
    getCount() {
      return count;
    }
  };
}
const counter = createCounter();
console.log(counter.increment()); // 1
console.log(counter.increment()); // 2
console.log(counter.decrement()); // 1
console.log(counter.count);       // undefined (Encapsulated!)

Problem 2.2: Fix the setTimeout in a loop problem.
-----------------------------------------
// Incorrect (prints 4 three times after 100ms):
for (var i = 1; i <= 3; i++) {
  setTimeout(() => console.log(i), 100);
}

// Fix 1: Use block-scoped 'let' (recommended)
for (let i = 1; i <= 3; i++) {
  setTimeout(() => console.log(i), 100); // Prints 1, 2, 3
}

// Fix 2: Use an IIFE to create a new scope closure
for (var i = 1; i <= 3; i++) {
  ((index) => {
    setTimeout(() => console.log(index), 100); // Prints 1, 2, 3
  })(i);
}

[x] JS-03: The "this" Keyword & Bind/Call/Apply
--------------------------------------------------------------------------------
💡 THEORY:
- 'this' points to the object executing the current function.
- In regular functions, 'this' depends on HOW the function is called.
- Arrow functions do NOT have their own 'this'. They inherit it lexically.
- call/apply: Immediately invoke the function with a specified 'this' value. 
  'call' takes comma-separated arguments; 'apply' takes arguments as an array.
- bind: Returns a new function with the bound 'this' context for later execution.

💻 POLYFILL IMPLEMENTATIONS:

Polyfill for call():
-----------------------------------------
Function.prototype.myCall = function(context = globalThis, ...args) {
  if (typeof this !== "function") {
    throw new TypeError("It must be called on a function");
  }
  
  // Wrap primitive values in objects (e.g. String, Number)
  context = context === null || context === undefined ? globalThis : Object(context);
  
  const fnKey = Symbol("fnKey"); // Unique property key
  context[fnKey] = this;         // Link current function to context object
  
  const result = context[fnKey](...args); // Execute the function
  delete context[fnKey];         // Clean up key
  return result;
};

Polyfill for apply():
-----------------------------------------
Function.prototype.myApply = function(context = globalThis, argsArray) {
  if (typeof this !== "function") {
    throw new TypeError("It must be called on a function");
  }
  
  context = context === null || context === undefined ? globalThis : Object(context);
  
  if (argsArray && !Array.isArray(argsArray) && typeof argsArray !== "object") {
    throw new TypeError("CreateListFromArrayLike called on non-object");
  }
  
  const fnKey = Symbol("fnKey");
  context[fnKey] = this;
  
  const result = argsArray ? context[fnKey](...argsArray) : context[fnKey]();
  delete context[fnKey];
  return result;
};

Polyfill for bind():
-----------------------------------------
Function.prototype.myBind = function(context, ...boundArgs) {
  if (typeof this !== "function") {
    throw new TypeError("It must be called on a function");
  }
  
  const originalFn = this;
  
  return function(...newArgs) {
    return originalFn.apply(context, [...boundArgs, ...newArgs]);
  };
};

[x] JS-04: Prototypes & Prototypal Inheritance
--------------------------------------------------------------------------------
💡 THEORY:
- prototype: A property unique to functions/classes. It defines properties and 
  methods to be shared by all instances created by that function as a constructor.
- __proto__: A property present on all instantiated objects that links to their 
  internal [[Prototype]] parent.
- Prototypal Chain: When trying to access a property, JS goes up the __proto__ 
  chain until it finds the property or hits null (Object.prototype.__proto__ is null).

💻 POLYFILL IMPLEMENTATIONS:

Polyfill for Object.create():
-----------------------------------------
Object.myCreate = function(proto, propertiesObject) {
  if (typeof proto !== "object" && typeof proto !== "function" && proto !== null) {
    throw new TypeError("Object prototype may only be an Object or null: " + proto);
  }
  
  // 1. Create a dummy constructor
  function F() {}
  
  // 2. Set prototype to the desired parent object
  F.prototype = proto;
  
  // 3. Create instance
  const obj = new F();
  
  // 4. Handle null prototype explicitly
  if (proto === null) {
    Object.setPrototypeOf(obj, null);
  }
  
  // 5. Define descriptors if passed
  if (propertiesObject !== undefined) {
    Object.defineProperties(obj, propertiesObject);
  }
  
  return obj;
};

[x] JS-05: Event Loop, Microtasks & Macrotasks
--------------------------------------------------------------------------------
💡 THEORY:
- Microtask Queue (higher priority): Promises (.then/.catch/finally), 
  queueMicrotask(), MutationObserver.
- Macrotask Queue (lower priority): setTimeout, setInterval, setImmediate, I/O.
- Node-specific: process.nextTick() executes immediately after the current 
  operation completes, before standard microtasks.

💻 OUTPUT QUESTION & RESOLUTION TRACE:

Predict the output:
-----------------------------------------
console.log("A"); 

setTimeout(() => console.log("B"), 0);

Promise.resolve().then(() => {
  console.log("C");
  setTimeout(() => console.log("D"), 0);
});

process.nextTick(() => console.log("E"));

console.log("F");

// 1. Synchronous logs: "A", "F"
// 2. process.nextTick executes: "E"
// 3. Microtask queue executes Promise: "C" (adds a setTimeout "D" to macrotasks)
// 4. Macrotask queue executes in FIFO order: "B", then "D"
// Output: A -> F -> E -> C -> B -> D

[x] JS-06: Promises & Async/Await
--------------------------------------------------------------------------------
💡 THEORY:
- Promises have 3 states: Pending, Fulfilled, Rejected.
- Async/await is syntactic sugar over promises.

💻 POLYFILL IMPLEMENTATIONS:

Promise.all(): Resolves when all resolve; rejects if any rejects.
-----------------------------------------
Promise.myAll = function(promises) {
  return new Promise((resolve, reject) => {
    if (!Array.isArray(promises)) {
      return reject(new TypeError("Arguments must be an array"));
    }
    const results = [];
    let completedCount = 0;
    const length = promises.length;
    
    if (length === 0) return resolve(results);
    
    promises.forEach((p, idx) => {
      Promise.resolve(p)
        .then(val => {
          results[idx] = val;
          completedCount++;
          if (completedCount === length) {
            resolve(results);
          }
        })
        .catch(reject);
    });
  });
};

Promise.allSettled(): Resolves when all promises settle (either resolve or reject).
-----------------------------------------
Promise.myAllSettled = function(promises) {
  return new Promise((resolve, reject) => {
    if (!Array.isArray(promises)) {
      return reject(new TypeError("Arguments must be an array"));
    }
    const results = [];
    let completedCount = 0;
    const length = promises.length;
    
    if (length === 0) return resolve(results);
    
    promises.forEach((p, idx) => {
      Promise.resolve(p)
        .then(value => {
          results[idx] = { status: "fulfilled", value };
        })
        .catch(reason => {
          results[idx] = { status: "rejected", reason };
        })
        .finally(() => {
          completedCount++;
          if (completedCount === length) {
            resolve(results);
          }
        });
    });
  });
};

Promise.race(): Resolves/Rejects with the first settled promise.
-----------------------------------------
Promise.myRace = function(promises) {
  return new Promise((resolve, reject) => {
    if (!Array.isArray(promises)) {
      return reject(new TypeError("Arguments must be an array"));
    }
    promises.forEach(p => {
      Promise.resolve(p).then(resolve, reject);
    });
  });
};

Promise.any(): Resolves with first fulfilled promise; rejects only if all fail.
-----------------------------------------
Promise.myAny = function(promises) {
  return new Promise((resolve, reject) => {
    if (!Array.isArray(promises)) {
      return reject(new TypeError("Arguments must be an array"));
    }
    const errors = [];
    let rejectedCount = 0;
    const length = promises.length;
    
    if (length === 0) {
      return reject(new AggregateError(errors, "All promises were rejected"));
    }
    
    promises.forEach((p, idx) => {
      Promise.resolve(p)
        .then(resolve)
        .catch(err => {
          errors[idx] = err;
          rejectedCount++;
          if (rejectedCount === length) {
            reject(new AggregateError(errors, "All promises were rejected"));
          }
        });
    });
  });
};

[x] JS-07: Array Polyfills & Methods
--------------------------------------------------------------------------------
💻 POLYFILL IMPLEMENTATIONS:

Array.prototype.map():
-----------------------------------------
Array.prototype.myMap = function(callback, thisArg) {
  if (this == null) throw new TypeError("Array.prototype.map called on null or undefined");
  if (typeof callback !== "function") throw new TypeError(callback + " is not a function");
  
  const O = Object(this);
  const len = O.length >>> 0;
  const A = new Array(len);
  
  for (let i = 0; i < len; i++) {
    if (i in O) {
      A[i] = callback.call(thisArg, O[i], i, O);
    }
  }
  return A;
};

Array.prototype.filter():
-----------------------------------------
Array.prototype.myFilter = function(callback, thisArg) {
  if (this == null) throw new TypeError("Array.prototype.filter called on null or undefined");
  if (typeof callback !== "function") throw new TypeError(callback + " is not a function");
  
  const O = Object(this);
  const len = O.length >>> 0;
  const res = [];
  
  for (let i = 0; i < len; i++) {
    if (i in O) {
      if (callback.call(thisArg, O[i], i, O)) {
        res.push(O[i]);
      }
    }
  }
  return res;
};

Array.prototype.reduce():
-----------------------------------------
Array.prototype.myReduce = function(callback, initialValue) {
  if (this == null) throw new TypeError("Array.prototype.reduce called on null or undefined");
  if (typeof callback !== "function") throw new TypeError(callback + " is not a function");
  
  const O = Object(this);
  const len = O.length >>> 0;
  
  let accumulator = initialValue;
  let k = 0;
  
  // If no initial value is provided, take the first element in the array
  if (accumulator === undefined) {
    let found = false;
    while (k < len) {
      if (k in O) {
        accumulator = O[k];
        found = true;
        k++;
        break;
      }
      k++;
    }
    if (!found) {
      throw new TypeError("Reduce of empty array with no initial value");
    }
  }
  
  for (; k < len; k++) {
    if (k in O) {
      accumulator = callback(accumulator, O[k], k, O);
    }
  }
  return accumulator;
};

Array.prototype.flat():
-----------------------------------------
Array.prototype.myFlat = function(depth = 1) {
  if (this == null) throw new TypeError("Array.prototype.flat called on null or undefined");
  
  const flatten = (arr, currentDepth) => {
    if (currentDepth <= 0) return arr.slice();
    
    return arr.reduce((acc, val) => {
      if (Array.isArray(val)) {
        acc.push(...flatten(val, currentDepth - 1));
      } else {
        acc.push(val);
      }
      return acc;
    }, []);
  };
  
  return flatten(this, depth);
};

[x] JS-08: Currying and Infinite Currying
--------------------------------------------------------------------------------
💡 THEORY:
- Currying is a technique of translating a function that takes multiple arguments 
  into a sequence of functions that each take a single argument. E.g., f(a, b, c) -> f(a)(b)(c).

💻 CODING SOLUTIONS:

Problem 8.1: Implement standard Currying utility
-----------------------------------------
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    } else {
      return function(...nextArgs) {
        return curried.apply(this, [...args, ...nextArgs]);
      };
    }
  };
}
// Example usage:
const sum = (a, b, c) => a + b + c;
const curriedSum = curry(sum);
console.log(curriedSum(1)(2)(3)); // 6
console.log(curriedSum(1, 2)(3)); // 6

Problem 8.2: Implement Infinite Currying: add(1)(2)(3)...(n)()
-----------------------------------------
function add(x) {
  return function(y) {
    if (y !== undefined) {
      return add(x + y); // keep returning function
    }
    return x; // return final sum when empty arg is called
  };
}
console.log(add(1)(2)(3)(4)()); // 10

[x] JS-09: Debounce and Throttle
--------------------------------------------------------------------------------
💡 THEORY:
- Debounce: Delay the execution of a function until a certain amount of time has 
  passed since the last call. (Good for Autocomplete/Search input, window resize).
- Throttle: Execute the function at most once within a specified time interval. 
  (Good for Scroll tracking, resize events, button double-click spamming).

💻 CODING SOLUTIONS:

Debounce implementation:
-----------------------------------------
function debounce(func, delay) {
  let timerId;
  return function(...args) {
    const context = this;
    clearTimeout(timerId); // Clear previous timer
    timerId = setTimeout(() => {
      func.apply(context, args);
    }, delay);
  };
}

Throttle implementation:
-----------------------------------------
function throttle(func, limit) {
  let inThrottle = false;
  return function(...args) {
    const context = this;
    if (!inThrottle) {
      func.apply(context, args);
      inThrottle = true;
      setTimeout(() => {
        inThrottle = false;
      }, limit);
    }
  };
}

[x] JS-10: Deep Copy vs Shallow Copy
--------------------------------------------------------------------------------
💡 THEORY:
- Shallow Copy: Copies only the top-level values. Nested objects/arrays are copied 
  by reference (pointing to the same memory location). 
  E.g. {...obj}, Object.assign({}, obj).
- Deep Copy: Recursively copies all properties, ensuring nested objects/arrays 
  are completely new instances in memory.

💻 CODING SOLUTIONS:

Problem 10.1: Implement custom deepClone function
-----------------------------------------
function deepClone(obj) {
  // Handle primitives, null, and undefined
  if (obj === null || typeof obj !== "object") {
    return obj;
  }

  // Handle Date
  if (obj instanceof Date) {
    return new Date(obj.getTime());
  }

  // Handle RegExp
  if (obj instanceof RegExp) {
    return new RegExp(obj);
  }

  // Handle Arrays
  if (Array.isArray(obj)) {
    const arrCopy = [];
    for (let i = 0; i < obj.length; i++) {
      arrCopy[i] = deepClone(obj[i]);
    }
    return arrCopy;
  }

  // Handle Objects
  const objCopy = {};
  for (let key in obj) {
    if (obj.hasOwnProperty(key)) {
      objCopy[key] = deepClone(obj[key]);
    }
  }
  
  return objCopy;
}

[x] MC-06: Task Queue with Concurrency Limit
--------------------------------------------------------------------------------
💡 MACHINE CODING SCENARIO:
Implement a class or function utility to execute asynchronous tasks with a 
concurrency limit (e.g. at most 3 tasks running in parallel). When a task finishes,
the next pending task starts automatically.

💻 CODING SOLUTIONS:

Class-based Task Queue with Concurrency Limit:
-----------------------------------------
class TaskQueue {
  constructor(concurrencyLimit) {
    this.concurrencyLimit = concurrencyLimit;
    this.queue = [];
    this.runningCount = 0;
  }

  // Enqueue a task. A task must be a function that returns a Promise.
  addTask(task) {
    return new Promise((resolve, reject) => {
      this.queue.push({ task, resolve, reject });
      this.runNext();
    });
  }

  runNext() {
    // Check if we can run another task
    if (this.runningCount >= this.concurrencyLimit || this.queue.length === 0) {
      return;
    }

    const { task, resolve, reject } = this.queue.shift();
    this.runningCount++;

    task()
      .then(resolve)
      .catch(reject)
      .finally(() => {
        this.runningCount--;
        this.runNext(); // Attempt to start the next task
      });
  }
}

// Example usage:
const tq = new TaskQueue(2); // Maximum 2 active tasks

const makeTask = (id, duration) => () => {
  console.log(`Task ${id} started`);
  return new Promise(res => setTimeout(() => {
    console.log(`Task ${id} finished`);
    res(id);
  }, duration));
};

tq.addTask(makeTask(1, 100));
tq.addTask(makeTask(2, 200));
tq.addTask(makeTask(3, 50)); // Should wait until Task 1 finishes (100ms)
tq.addTask(makeTask(4, 80));
